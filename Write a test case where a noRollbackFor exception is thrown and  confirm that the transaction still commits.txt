MinorWarningException.java :
package com.lab; 
public class MinorWarningException extends RuntimeException { 
    public MinorWarningException(String msg) { super(msg); } 
} 


OrderService.java :
package com.lab; 
 
import org.springframework.stereotype.Service; 
import org.springframework.transaction.annotation.Transactional; 
 
@Service 
public class OrderService { 
    @Transactional(noRollbackFor = MinorWarningException.class) 
    public void placeOrder() { 
        System.out.println("DML: Order record inserted into database 
successfully."); 
        throw new MinorWarningException("Notification server unreachable 
(Non-critical)."); 
    } 
}

<beans xmlns="http://www.springframework.org/schema/beans"

http://www.springframework.org/schema/context/spring-context.xsd

http://www.springframework.org/schema/tx/spring-tx.xsd"> 
 
    <context:component-scan base-package="com.lab"/>

class="org.springframework.jdbc.datasource.DriverManagerDataSource"> 
        <property name="driverClassName" value="org.h2.Driver"/> 
        <property name="url" value="jdbc:h2:mem:bankdb;DB_CLOSE_DELAY=

    <bean id="transactionManager" 
class="org.springframework.jdbc.datasource.DataSourceTransactionManager"> 
        <property name="dataSource" ref="dataSource"/> 
    </bean> 
    <tx:annotation-driven transaction-manager="transactionManager"/> 
</beans> 


App.java :
package com.lab; 
import org.springframework.context.ApplicationContext; 
import 
org.springframework.context.support.ClassPathXmlApplicationContext; 
 
public class App { 
    public static void main(String[] args) { 
        ApplicationContext ctx = new 
ClassPathXmlApplicationContext("Beans.xml"); 
        OrderService os = ctx.getBean(OrderService.class); 
        try { 
            os.placeOrder(); 
        } catch (MinorWarningException ex) { 
            System.out.println("Caught Expected Warning: " + 
ex.getMessage()); 
            System.out.println("Transaction Outcome: COMMITTED 
(noRollbackFor rule active)."); 
        } 
        ((ClassPathXmlApplicationContext) ctx).close(); 
    } 
}
