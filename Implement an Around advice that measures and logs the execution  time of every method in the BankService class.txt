package com.lab; 
import org.springframework.stereotype.Service; 
@Service 
public class BankService { 
    public void deposit(double amount) { 
        try { Thread.sleep(65); } catch (InterruptedException ignored) {} 
        System.out.println("Deposited amount: Rs. " + amount); 
    } 
} 


TimingAspect.java :
package com.lab; 
 
import org.aspectj.lang.ProceedingJoinPoint; 
import org.aspectj.lang.annotation.Around; 
import org.aspectj.lang.annotation.Aspect; 
import org.springframework.stereotype.Component; 
 
@Aspect 
@Component 
public class TimingAspect { 
    @Around("execution(* com.lab.BankService.*(..))") 
    public Object profile(ProceedingJoinPoint pjp) throws Throwable { 
        long start = System.currentTimeMillis(); 
        Object res = pjp.proceed(); 
        long duration = System.currentTimeMillis() - start; 
        System.out.println("[PROFILER] Method '" + 
pjp.getSignature().getName() + "' executed in " + duration + " ms"); 
        return res; 
    } 
}

package com.lab; 
import org.springframework.context.ApplicationContext; 

import 
org.springframework.context.support.ClassPathXmlApplicationContext; 
 
public class App { 
    public static void main(String[] args) { 
        ApplicationContext ctx = new 
ClassPathXmlApplicationContext("Beans.xml"); 
        BankService bank = ctx.getBean(BankService.class); 
        bank.deposit(50000.0); 
        ((ClassPathXmlApplicationContext) ctx).close(); 
    } 
}
