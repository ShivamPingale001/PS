GetterAspect.java :
package com.lab; 
import org.aspectj.lang.JoinPoint; 
import org.aspectj.lang.annotation.Aspect; 
import org.aspectj.lang.annotation.Before; 
import org.springframework.stereotype.Component; 
 
@Aspect 
@Component 
public class GetterAspect { 
    @Before("execution(* com.lab.*.get*(..))") 
    public void beforeGetter(JoinPoint jp) { 
        System.out.println("[AUDIT LOG] Intercepted query call -> " + 
jp.getSignature().toShortString()); 
    } 
}

App.java :
package com.lab; 
 
import org.springframework.context.ApplicationContext; 
import 
org.springframework.context.support.ClassPathXmlApplicationContext; 
 
public class App { 
    public static void main(String[] args) { 
        ApplicationContext ctx = new 
ClassPathXmlApplicationContext("Beans.xml"); 
        AccountService acc = ctx.getBean(AccountService.class); 
        acc.getAccountHolder(); 
        acc.getBalance(); 
        acc.modifyPin(); 
        ((ClassPathXmlApplicationContext) ctx).close(); 
    } 
}
