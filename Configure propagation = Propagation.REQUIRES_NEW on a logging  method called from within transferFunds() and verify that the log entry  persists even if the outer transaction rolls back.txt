AuditLogService.java :
package com.lab; 
 
import org.springframework.stereotype.Service; 
import org.springframework.transaction.annotation.Propagation; 
import org.springframework.transaction.annotation.Transactional; 
 
@Service 
public class AuditLogService { 
    @Transactional(propagation = Propagation.REQUIRES_NEW) 
    public void logAttempt(String message) { 
        System.out.println("[REQUIRES_NEW] Audit log saved & committed 
independently: " + message); 
    } 
} 

 
TransferService.java :
package com.lab; 
 
import org.springframework.beans.factory.annotation.Autowired; 
import org.springframework.stereotype.Service; 
import org.springframework.transaction.annotation.Transactional; 
 
@Service 
public class TransferService { 
    @Autowired 
    private AuditLogService logService; 
 
    @Transactional 
    public void transferFunds(double amount) { 
        logService.logAttempt("Transfer attempt initiated for Rs. " + 
amount); 
        if (amount > 50000) { 
            throw new RuntimeException("Limit exceeded! Outer fund 
transfer transaction rolled back."); 
        } 
        System.out.println("Funds transferred successfully."); 
    } 
}

App.java :
package com.lab; 
 
import org.springframework.context.ApplicationContext; 
import 
org.springframework.context.support.ClassPathXmlApplicationContext; 
 
public class App { 
    public static void main(String[] args) { 
        ApplicationContext ctx = new 
ClassPathXmlApplicationContext("Beans.xml"); 
        TransferService ts = ctx.getBean(TransferService.class); 
        try { 
            ts.transferFunds(75000); 
        } catch (Exception e) { 
            System.err.println("Caller Caught Rollback: " + 
e.getMessage()); 
        } 
        ((ClassPathXmlApplicationContext) ctx).close(); 
    } 
}
